# =======================================mysql-8
docker run -d \
  --name mysql \
  --network lli-net \
  -p 3306:3306 \
  -e MYSQL_ROOT_PASSWORD=123456 \
  -e TZ=Asia/Shanghai \
  mysql:8.4.6 \
  --character-set-server=utf8mb4 \
  --collation-server=utf8mb4_unicode_ci

# =======================================nacos-standalone-derby
docker run --name nacos-standalone-derby \
  --network lli-net \
  -e MODE=standalone \
  -e TZ=Asia/Shanghai \
  -e NACOS_AUTH_ENABLE=true \
  -e NACOS_AUTH_TOKEN=$(echo -n "my-nacos-secret-key-at-least-32-chars" | base64) \
  -e NACOS_AUTH_IDENTITY_KEY=lli \
  -e NACOS_AUTH_IDENTITY_VALUE=lli \
  -p 8848:8848 \
  -p 9848:9848 \
  -d nacos/nacos-server:v2.5.4

docker run -d --name seata-server \
  --network lli-net \
  -p 8091:8091 -p 7091:7091 \
  -e TZ=Asia/Shanghai \
  -e SEATA_IP=<宿主机IP> \
  -v /Users/lihao/Seata-Volume/resources:/seata-server/resources \
  -v /Users/lihao/Seata-Volume/libs:/seata-server/libs \
  -v /etc/localtime:/etc/localtime:ro \
  apache/seata-server:2.5.0
